### 集成到现有项目

#### 在main.go中集成中间件

```go
package main

import (
    "qingyu/pkg/logger"
    "qingyu/internal/middleware/builtin"
    "qingyu/internal/middleware/ratelimit"
    "qingyu/pkg/metrics"
)

func main() {
    // 1. 初始化日志
    logger.Init(&logger.Config{
        Level:  "info",
        Format: "json",
        Output: "stdout",
    })

    // 2. 创建路由
    router := gin.Default()

    // 3. 应用中间件（顺序很重要）
    requestIDMiddleware := builtin.NewRequestIDMiddleware()
    recoveryMiddleware := builtin.NewRecoveryMiddleware(logger.Get().Logger)
    loggerMiddleware := builtin.NewLoggerMiddleware(logger.Get().Logger)
    errorHandlerMiddleware := builtin.NewErrorHandlerMiddleware(logger.Get().Logger)
    corsMiddleware := builtin.NewCORSMiddleware()

    rateLimitConfig := &ratelimit.RateLimitConfig{
        Enabled:  true,
        Strategy: "token_bucket",
        Rate:     10,
        Burst:    100,
    }
    rateLimitMiddleware, _ := ratelimit.NewRateLimitMiddleware(rateLimitConfig, logger.Get().Logger)

    router.Use(
        requestIDMiddleware.Handler(),             // 请求ID
        recoveryMiddleware.Handler(),              // 异常恢复
        loggerMiddleware.Handler(),                // 日志记录
        metrics.Middleware(),                      // Prometheus监控
        rateLimitMiddleware.Handler(),             // 限流
        errorHandlerMiddleware.Handler(),          // 错误处理
        corsMiddleware.Handler(),                  // CORS
    )

    // 4. 注册健康检查和监控端点
    healthAPI := system.NewHealthAPI(db, redis)
    router.GET("/health", healthAPI.GetHealth)
    router.GET("/health/live", healthAPI.Liveness)
    router.GET("/health/ready", healthAPI.Readiness)
    router.GET("/metrics", metrics.MetricsHandler())

    // 5. 启动服务
    router.Run(":8080")
}
```
