```go
func TestDocumentVersionCalculation(t *testing.T) {
    now := time.Now()
    doc := &writer.Document{
        CreatedAt: now.Add(-25 * time.Minute),
        UpdatedAt: now,
    }
    
    status, err := service.GetSaveStatus(ctx, doc.ID)
    assert.Equal(t, 6, status.CurrentVersion) // 25/5 + 1 = 6
}
```
